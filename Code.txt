/**
 * ============================================================================
 * 1. PUNTO DE ENTRADA PRINCIPAL (WEBHOOK + CHAT COACH MBE)
 * ============================================================================
 */
function doPost(e) {
  let isCoach = false;
  try { isCoach = JSON.parse(e.postData.contents).action === 'ask_coach'; } catch (_) {}
  if (isCoach) return doPostInner_(e);

  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(25000);
  } catch (_) {
    return jsonResponse_({ status: 'error', message: 'Servidor ocupado, reintenta' });
  }
  try {
    return doPostInner_(e);
  } finally {
    lock.releaseLock();
  }
}

function doPostInner_(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse_({ status: 'error', message: 'Payload vacío' });
    }

    const payload = JSON.parse(e.postData.contents);

    // Si la petición viene del botón "🧠 Coach" dentro de la app ARGON
    if (payload && payload.action === 'ask_coach') {
      return handleArgonCoachChat_(payload);
    }

    // Procesamiento normal de envíos de HC Webhook (Health Connect)
    const parsedByDate = extractHealthConnectByDate_(payload);
    const detectedDates = Object.keys(parsedByDate).sort();
    const syncedSummaries = [];

    detectedDates.forEach(dateStr => {
      const dayComputed = upsertAndComputeDayInSupabase_(dateStr, parsedByDate[dateStr]);
      syncedSummaries.push({
        log_date: dateStr,
        weight_kg: dayComputed.weight_kg,
        body_fat_pct: dayComputed.body_fat_pct,
        steps: dayComputed.steps_count,
        intake_kcal: dayComputed.intake_kcal,
        prot_g: dayComputed.intake_protein_g,
        carbs_g: dayComputed.intake_carbs_g,
        fat_g: dayComputed.intake_fat_g,
        strain: dayComputed.strain_score,
        recovery: dayComputed.recovery_score
      });
    });

    try { logWebhookAudit_(payload, detectedDates, syncedSummaries); } catch (le) { Logger.log('Audit: ' + le.message); }
    syncExpedienteToGoogleDrive_();

    return jsonResponse_({
      status: 'ok',
      engine: 'ARGON_v6.0_PERSISTENT_MEALS',
      days_updated: detectedDates.length,
      summary: syncedSummaries
    });
  } catch (err) {
    Logger.log('Error en doPost: ' + err.stack);
    return jsonResponse_({ status: 'error', message: err.message });
  }
}

/**
 * ============================================================================
 * 2. PARSER DE FECHAS (CARACAS UTC-4) Y ESTIMADOR DE MACROS
 * ============================================================================
 */
function extractRawTimestamp_(rec) {
  if (!rec || typeof rec !== 'object') return null;
  const candidates = [
    rec.start_time, rec.startTime, rec.time, rec.timestamp,
    rec.date_time, rec.dateTime, rec.log_date, rec.date,
    rec.end_time, rec.endTime, rec.created_at,
    rec.metadata && (
      rec.metadata.start_time || rec.metadata.startTime ||
      rec.metadata.time || rec.metadata.last_modified_time ||
      rec.metadata.lastModifiedTime
    )
  ];
  for (let i = 0; i < candidates.length; i++) {
    if (candidates[i] !== undefined && candidates[i] !== null && candidates[i] !== '') {
      return candidates[i];
    }
  }
  const keys = Object.keys(rec);
  for (let j = 0; j < keys.length; j++) {
    const k = keys[j].toLowerCase();
    if ((k.includes('time') || k.includes('date')) && typeof rec[keys[j]] !== 'object') {
      return rec[keys[j]];
    }
  }
  return null;
}

function toCaracasDateAndMinutes_(rawTs) {
  if (rawTs === null || rawTs === undefined || rawTs === '') return null;

  if (typeof rawTs === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(rawTs.trim())) {
    const dStr = rawTs.trim();
    return { dateStr: dStr, epochMs: new Date(dStr + 'T12:00:00-04:00').getTime() };
  }

  let dt = null;
  if (typeof rawTs === 'number' || (typeof rawTs === 'string' && /^\d+(\.\d+)?$/.test(rawTs.trim()))) {
    const num = parseFloat(rawTs);
    const ms = num < 100000000000 ? num * 1000 : num;
    dt = new Date(ms);
  } else if (typeof rawTs === 'string') {
    let norm = rawTs.trim();
    if (/^\d{4}-\d{2}-\d{2}\s+\d{2}:\d{2}/.test(norm)) norm = norm.replace(/\s+/, 'T');
    dt = new Date(norm);
  }

  if (!dt || isNaN(dt.getTime())) return null;

  return {
    dateStr: Utilities.formatDate(dt, ARGON_CONFIG.TIMEZONE, 'yyyy-MM-dd'),
    epochMs: dt.getTime()
  };
}

function estimateMacrosIfZero_(foodName, kcal, prot, carbs, fat) {
  if (prot > 0 || carbs > 0 || fat > 0 || !kcal || kcal <= 0) {
    return { prot: +(prot || 0).toFixed(1), carbs: +(carbs || 0).toFixed(1), fat: +(fat || 0).toFixed(1) };
  }
  const n = String(foodName || '').toLowerCase();

  // Perfiles de macros por categoría (proteína, carbs, grasa) - suman 1.0
  const profiles = [
    { regex: /pollo|pechuga|pavo|atun|atún|pescado|clara|whey/,       p: 0.70, c: 0.05, f: 0.25, weight: 1.0 },
    { regex: /carne|bistec|res|cerdo|chuleta|lomito|jamon|jamón/,       p: 0.48, c: 0.02, f: 0.50, weight: 1.0 },
    { regex: /queso|huevo|aguacate|mantequilla/,                        p: 0.32, c: 0.04, f: 0.64, weight: 1.0 },
    { regex: /arroz|platano|plátano|arepa|waffle|yuca|pan|pasta|avena|pimiento|ensalada|pure|puré|papa|patata/, p: 0.12, c: 0.50, f: 0.38, weight: 1.0 },
    { regex: /nestea|jugo|batido|coca|refresco|azucar/,                 p: 0.02, c: 0.96, f: 0.02, weight: 1.0 },
    { regex: /empanada|frita|chucheria|chuchería|chocolate/,            p: 0.12, c: 0.46, f: 0.42, weight: 1.0 }
  ];

  // Acumulación ponderada: platos combinados (ej. "pollo con arroz") promedian ambos perfiles
  let totalWeight = 0;
  let accP = 0, accC = 0, accF = 0;
  profiles.forEach(profile => {
    if (profile.regex.test(n)) {
      accP += profile.p * profile.weight;
      accC += profile.c * profile.weight;
      accF += profile.f * profile.weight;
      totalWeight += profile.weight;
    }
  });

  // Si no matchea ningún perfil, usa default balanceado
  if (totalWeight === 0) {
    accP = 0.25; accC = 0.45; accF = 0.30;
    totalWeight = 1;
  }

  const pRatio = accP / totalWeight;
  const cRatio = accC / totalWeight;
  const fRatio = accF / totalWeight;

  return {
    prot: Math.round((kcal * pRatio) / 4),
    carbs: Math.round((kcal * cRatio) / 4),
    fat: Math.round((kcal * fRatio) / 9)
  };
}

/**
 * ============================================================================
 * 3. EXTRACTOR MULTICATEGORÍA DE HEALTH CONNECT POR FECHA
 * ============================================================================
 */
function extractHealthConnectByDate_(payload) {
  const daysMap = {};
  const todayCaracas = Utilities.formatDate(new Date(), ARGON_CONFIG.TIMEZONE, 'yyyy-MM-dd');

  function getDayBucket(dateStr) {
    const d = (dateStr && /^\d{4}-\d{2}-\d{2}/.test(dateStr)) ? dateStr.substring(0, 10) : todayCaracas;
    if (!daysMap[d]) {
      daysMap[d] = {
        log_date: d,
        has_steps_payload: false,
        steps_map: {},
        has_nutrition_payload: false,
        meals_map: {},
        rhr_list: [],
        rmssd_list: [],
        resp_list: [],
        spo2_list: [],
        latest_weight_kg: null,
        latest_weight_ts: 0,
        latest_bf_pct: null,
        latest_bf_ts: 0,
        tmb_kcal: null,
        sleep_total_min: null,
        sleep_deep_min: null,
        sleep_rem_min: null,
        sleep_light_min: null,
        sleep_tib_min: null,
        workout_map: {},
        nap_map: {}
      };
    }
    return daysMap[d];
  }

  function extractAnyNumber(obj, preferredKeys) {
    if (obj === null || obj === undefined) return null;
    if (typeof obj === 'number') return obj;
    if (typeof obj === 'string') {
      const n = parseFloat(obj.replace(',', '.'));
      return isNaN(n) ? null : n;
    }
    if (typeof obj === 'object') {
      for (let i = 0; i < preferredKeys.length; i++) {
        const k = preferredKeys[i];
        if (obj[k] !== undefined && obj[k] !== null) {
          const val = extractAnyNumber(obj[k], preferredKeys);
          if (val !== null) return val;
        }
      }
      if (obj.inKilojoules !== undefined) return parseFloat(obj.inKilojoules) / 4.184;
      if (obj.inJoules !== undefined) return parseFloat(obj.inJoules) / 4184.0;
      if (obj.inGrams !== undefined && preferredKeys.includes('inKilograms')) return parseFloat(obj.inGrams) / 1000.0;
      const keys = Object.keys(obj);
      for (let j = 0; j < keys.length; j++) {
        const kName = keys[j].toLowerCase();
        if (kName.includes('time') || kName.includes('date') || kName.includes('id') || kName.includes('type') || kName.includes('zone')) continue;
        const subVal = obj[keys[j]];
        if (typeof subVal === 'number' && !isNaN(subVal)) return subVal;
      }
    }
    return null;
  }

  function processRecord(rec, categoryName) {
    if (!rec || typeof rec !== 'object') return;
    const cat = String(categoryName || rec.type || rec.record_type || rec.recordType || '').toLowerCase();

    const rawTs = extractRawTimestamp_(rec);
    const parsedTs = toCaracasDateAndMinutes_(rawTs);
    if (!parsedTs) return;

    const { dateStr, epochMs } = parsedTs;
    const bucket = getDayBucket(dateStr);

    // 1. PESO
    if (cat === 'weight' || cat.includes('weight') || rec.weight !== undefined || rec.weight_kg !== undefined) {
      let w = extractAnyNumber(
        rec.weight !== undefined ? rec.weight : (rec.weight_kg !== undefined ? rec.weight_kg : (rec.mass !== undefined ? rec.mass : rec)),
        ['inKilograms', 'in_kilograms', 'kilograms', 'kg', 'value', 'mass']
      );
      if (w && w > 35000) w = w / 1000.0;
      if (w && w >= 40 && w <= 220 && epochMs >= bucket.latest_weight_ts) {
        bucket.latest_weight_kg = +w.toFixed(2);
        bucket.latest_weight_ts = epochMs;
      }
    }

    // 2. GRASA CORPORAL
    if (cat === 'body_fat' || cat.includes('bodyfat') || cat.includes('body_fat') || rec.body_fat_pct !== undefined) {
      let bf = extractAnyNumber(
        rec.percentage !== undefined ? rec.percentage : (rec.body_fat !== undefined ? rec.body_fat : (rec.bodyFat !== undefined ? rec.bodyFat : rec)),
        ['inPercent', 'in_percent', 'percent', 'percentage', 'value']
      );
      if (bf && bf > 0 && bf < 1.0) bf = bf * 100.0;
      if (bf && bf >= 4 && bf <= 65 && epochMs >= bucket.latest_bf_ts) {
        bucket.latest_bf_pct = +bf.toFixed(1);
        bucket.latest_bf_ts = epochMs;
      }
    }

    // 3. PASOS
    if (cat === 'steps' || cat.includes('step') || rec.steps_count !== undefined) {
      bucket.has_steps_payload = true;
      const count = extractAnyNumber(rec.count !== undefined ? rec.count : (rec.steps_count !== undefined ? rec.steps_count : rec), ['count', 'steps', 'value']);
      if (count !== null && count >= 0 && count < 100000) {
        const stepKey = String(rec.id || (rec.metadata && rec.metadata.id) || epochMs);
        bucket.steps_map[stepKey] = Math.round(count);
      }
    }

    // 4. NUTRICIÓN ESTRICTA (Ignora active_calories / total_calories y evita colisión de timestamps)
    if ((cat === 'nutrition' || cat.includes('nutrition') || cat.includes('meal') || cat.includes('food')) && !cat.includes('calories')) {
      bucket.has_nutrition_payload = true;
      const nutSource = rec.nutrients || rec.macros || rec;
      const kcal = extractAnyNumber(rec.energy || rec.calories || rec.total_calories || rec.intake_kcal || rec.totalCalories || nutSource.energy || nutSource.calories, ['inKilocalories', 'in_kilocalories', 'kilocalories', 'kcal', 'value']);
      const rawProt = extractAnyNumber(rec.protein_grams || rec.protein || rec.proteins || rec.protein_g || rec.intake_protein_g || nutSource.protein || nutSource.proteins, ['inGrams', 'in_grams', 'grams', 'g', 'value']) || 0;
      const rawCarbs = extractAnyNumber(rec.carbs_grams || rec.total_carbohydrate || rec.totalCarbohydrate || rec.carbohydrates || rec.carbs || rec.carbs_g || rec.intake_carbs_g || nutSource.carbohydrates || nutSource.carbs, ['inGrams', 'in_grams', 'grams', 'g', 'value']) || 0;
      const rawFat = extractAnyNumber(rec.fat_grams || rec.total_fat || rec.totalFat || rec.fat || rec.fats || rec.fat_g || rec.intake_fat_g || nutSource.fat || nutSource.fats, ['inGrams', 'in_grams', 'grams', 'g', 'value']) || 0;

      const foodName = String(rec.name || rec.meal_name || rec.mealName || rec.food_name || rec.title || rec.meal_type || rec.mealType || 'Alimento').trim();
      const finalKcal = Math.round(kcal || ((rawProt * 4) + (rawCarbs * 4) + (rawFat * 9)));

      if (finalKcal > 0) {
        const macros = estimateMacrosIfZero_(foodName, finalKcal, rawProt, rawCarbs, rawFat);
        const recUid = rec.id || (rec.metadata && rec.metadata.id) || `${epochMs}_${finalKcal}`;
        const cleanName = foodName.replace(/\s+/g, '_');
        let mealId = `${dateStr}_${recUid}_${cleanName}_${finalKcal}`;
        if (!rec.id && !(rec.metadata && rec.metadata.id)) {
          const baseId = mealId;
          let n = 2;
          while (bucket.meals_map[mealId]) mealId = baseId + '#' + n++;
        }
        bucket.meals_map[mealId] = {
          id: mealId,
          date_caracas: dateStr,
          est: !(rawProt > 0 || rawCarbs > 0 || rawFat > 0),
          time_caracas: Utilities.formatDate(new Date(epochMs), ARGON_CONFIG.TIMEZONE, 'HH:mm'),
          kcal: finalKcal,
          prot: macros.prot,
          carbs: macros.carbs,
          fat: macros.fat,
          name: foodName
        };
      }
    }

    // 5. VFC (Mediana en webhook para neutralizar artefactos PPG) Y FC REPOSO
    if (cat.includes('variability') || cat.includes('hrv') || rec.rmssd_ms !== undefined || rec.heartRateVariabilityMillis !== undefined) {
      const hrv = extractAnyNumber(rec.rmssd_millis || rec.heart_rate_variability_millis || rec.heartRateVariabilityMillis || rec.rmssd_ms || rec.rmssd || rec.hrv || rec, ['inMilliseconds', 'ms', 'value']);
      if (hrv && hrv >= 10 && hrv <= 180) bucket.rmssd_list.push(hrv);
    }
    if (cat.includes('resting_heart') || cat.includes('restingheart') || cat === 'rhr' || rec.rhr_bpm !== undefined) {
      const rhr = extractAnyNumber(rec.beats_per_minute || rec.beatsPerMinute || rec.rhr_bpm || rec.bpm || rec, ['beatsPerMinute', 'beats_per_minute', 'bpm', 'value']);
      if (rhr && rhr >= 35 && rhr <= 110) bucket.rhr_list.push(rhr);
    }

    // 6. RESPIRACIÓN Y SPO2
    if (cat.includes('respiratory') || rec.resp_rate_brpm !== undefined) {
      const rr = extractAnyNumber(rec.rate || rec.resp_rate_brpm || rec, ['rate', 'brpm', 'value']);
      if (rr && rr >= 8 && rr <= 35) bucket.resp_list.push(rr);
    }
    if (cat.includes('oxygen') || cat.includes('spo2') || rec.spo2_pct !== undefined) {
      let sp = extractAnyNumber(rec.percentage || rec.spo2_pct || rec, ['inPercent', 'in_percent', 'percent', 'percentage', 'value']);
      if (sp && sp > 0 && sp <= 1.0) sp = sp * 100.0;
      if (sp && sp >= 80 && sp <= 100) bucket.spo2_list.push(sp);
    }

    // 7. SUEÑO (Conserva siempre el episodio principal >= 120m para no pisar la noche con siestas)
    if (cat === 'sleep' || cat.includes('sleep') || Array.isArray(rec.stages)) {
      const wakeRaw = rec.end_time || rec.endTime || rec.session_end_time || rawTs;
      const wakeParsed = toCaracasDateAndMinutes_(wakeRaw) || parsedTs;
      const sleepBucket = getDayBucket(wakeParsed.dateStr);

      if (Array.isArray(rec.stages) && rec.stages.length > 0) {
        let deep = 0, rem = 0, light = 0, awake = 0;
        rec.stages.forEach(st => {
          const stStart = toCaracasDateAndMinutes_(st.start_time || st.startTime);
          const stEnd = toCaracasDateAndMinutes_(st.end_time || st.endTime);
          if (!stStart || !stEnd) return;
          const durMin = Math.max(0, Math.round((stEnd.epochMs - stStart.epochMs) / 60000));
          const stType = String(st.stage || st.type || '').toLowerCase();
          if (stType === '5' || stType.includes('deep')) deep += durMin;
          else if (stType === '6' || stType.includes('rem')) rem += durMin;
          else if (stType === '1' || stType === '3' || stType === '7' || stType.includes('awake') || stType.includes('out_of_bed')) awake += durMin;
          else if (durMin > 0) light += durMin;
        });
        const totalSt = deep + rem + light;
        if (totalSt >= 10 && totalSt < 120) {
          const fs = toCaracasDateAndMinutes_(rec.stages[0].start_time || rec.stages[0].startTime);
          const hr = fs ? parseInt(Utilities.formatDate(new Date(fs.epochMs), ARGON_CONFIG.TIMEZONE, 'H'), 10) : -1;
          if (hr >= 8 && hr <= 20) sleepBucket.nap_map[String(fs.epochMs)] = totalSt;
        }
        if (totalSt >= 120 && totalSt >= (sleepBucket.sleep_total_min || 0)) {
          sleepBucket.sleep_deep_min = deep;
          sleepBucket.sleep_rem_min = rem;
          sleepBucket.sleep_light_min = light;
          sleepBucket.sleep_total_min = totalSt;
          sleepBucket.sleep_tib_min = totalSt + awake;
        }
      } else {
        const sStart = toCaracasDateAndMinutes_(rec.start_time || rec.startTime);
        const sEnd = toCaracasDateAndMinutes_(rec.end_time || rec.endTime);
        if (sStart && sEnd) {
          const durMin = Math.round((sEnd.epochMs - sStart.epochMs) / 60000);
          if (durMin >= 10 && durMin < 120) {
            const hr2 = parseInt(Utilities.formatDate(new Date(sStart.epochMs), ARGON_CONFIG.TIMEZONE, 'H'), 10);
            if (hr2 >= 8 && hr2 <= 20) sleepBucket.nap_map[String(sStart.epochMs)] = durMin;
          }
          if (durMin >= 120 && durMin <= 840 && durMin >= (sleepBucket.sleep_total_min || 0)) {
            sleepBucket.sleep_total_min = durMin;
            sleepBucket.sleep_tib_min = Math.round(durMin * 1.06);
          }
        }
      }
    }

    // 8. EJERCICIO
    if (cat === 'exercise' || cat.includes('exercise')) {
      const eStart = toCaracasDateAndMinutes_(rec.start_time || rec.startTime);
      const eEnd = toCaracasDateAndMinutes_(rec.end_time || rec.endTime);
      if (eStart && eEnd) {
        const wMin = Math.round((eEnd.epochMs - eStart.epochMs) / 60000);
        if (wMin >= 5 && wMin <= 300) {
          bucket.workout_map[String(rec.id || eStart.epochMs)] = wMin;
        }
      }
    }
  }

  if (payload && typeof payload === 'object' && !Array.isArray(payload)) {
    Object.keys(payload).forEach(topKey => {
      const kLower = topKey.toLowerCase();
      if (kLower === 'timestamp' || kLower === 'app_version' || kLower === 'device') return;
      const val = payload[topKey];
      if (Array.isArray(val)) {
        val.forEach(item => processRecord(item, kLower));
      } else if (val && typeof val === 'object') {
        if (Array.isArray(val.records)) val.records.forEach(item => processRecord(item, kLower));
        else if (Array.isArray(val.data)) val.data.forEach(item => processRecord(item, kLower));
        else processRecord(val, kLower);
      }
    });
  } else if (Array.isArray(payload)) {
    payload.forEach(item => processRecord(item, ''));
  }

  getDayBucket(todayCaracas);
  return daysMap;
}

/**
 * ============================================================================
 * 4. UPSERT CON ACUMULACIÓN PERSISTENTE EN EL DÍA Y REINICIO A LAS 00:00
 * ============================================================================
 */
function upsertAndComputeDayInSupabase_(dateStr, extractedDay) {
  const existingRows = supabaseGet_(`daily_raw_biometrics?log_date=eq.${dateStr}&limit=1`);
  const existing = existingRows.length > 0 ? existingRows[0] : {};
  if (parseInt(existing.total_sleep_min, 10) === 450 && parseInt(existing.deep_sleep_min, 10) === 84 && parseInt(existing.rem_sleep_min, 10) === 96) {
    existing.total_sleep_min = 0; existing.time_in_bed_min = 0; existing.deep_sleep_min = 0; existing.rem_sleep_min = 0; existing.light_sleep_min = 0;
  }

  const historyRows = supabaseGet_(`daily_raw_biometrics?log_date=lt.${dateStr}&order=log_date.desc&limit=14`);
  const lastKnownWeight = historyRows.length > 0 && historyRows[0].weight_kg ? parseFloat(historyRows[0].weight_kg) : 86.80;
  const lastKnownBf = historyRows.length > 0 && historyRows[0].body_fat_pct ? parseFloat(historyRows[0].body_fat_pct) : 24.4;

  const todayWeight = extractedDay.latest_weight_kg || (existing.weight_kg ? parseFloat(existing.weight_kg) : lastKnownWeight);
  const todayBf = extractedDay.latest_bf_pct || (existing.body_fat_pct ? parseFloat(existing.body_fat_pct) : lastKnownBf);

  const stepKeys = Object.keys(extractedDay.steps_map);
  let todaySteps = existing.steps_count || 0;
  if (stepKeys.length > 0) {
    const webhookSteps = stepKeys.reduce((sum, k) => sum + extractedDay.steps_map[k], 0);
    todaySteps = Math.max(todaySteps, webhookSteps);
  }

  const exKeys = Object.keys(extractedDay.workout_map);
  let todayWorkoutMin = existing.pickleball_or_gym_minutes || 0;
  if (exKeys.length > 0) {
    const webhookWork = exKeys.reduce((sum, k) => sum + extractedDay.workout_map[k], 0);
    todayWorkoutMin = Math.max(todayWorkoutMin, webhookWork);
  }

  // Conserva todas las comidas registradas hoy (tanto manuales como de Zepp previas)
  const mergedMealsMap = {};
  if (Array.isArray(existing.meals_log_json)) {
    existing.meals_log_json.forEach(m => {
      if (!m || !m.kcal) return;
      const belongsToday = (m.date_caracas === dateStr) ||
                           String(m.id || '').startsWith('manual_') ||
                           String(m.id || '').startsWith(dateStr);
      if (belongsToday) {
        const dedupKey = m.id || `${m.name}_${m.kcal}_${m.time_caracas || 'manual'}`;
        mergedMealsMap[dedupKey] = m;
      }
    });
  }

  Object.keys(extractedDay.meals_map).forEach(mId => {
    const mealObj = extractedDay.meals_map[mId];
    if (!mealObj.date_caracas || mealObj.date_caracas === dateStr) {
      const dedupKey = mealObj.id || `${mealObj.name}_${mealObj.kcal}_${mealObj.time_caracas || ''}`;
      mergedMealsMap[dedupKey] = mealObj;
    }
  });

  const mealsList = Object.keys(mergedMealsMap).map(k => mergedMealsMap[k]);

  const totalKcal = mealsList.reduce((acc, m) => acc + (parseInt(m.kcal, 10) || 0), 0);
  const totalProt = Math.round(mealsList.reduce((acc, m) => acc + (parseFloat(m.prot) || 0), 0));
  const totalCarbs = Math.round(mealsList.reduce((acc, m) => acc + (parseFloat(m.carbs) || 0), 0));
  const totalFat = Math.round(mealsList.reduce((acc, m) => acc + (parseFloat(m.fat) || 0), 0));

  const ffmPct = +(100.0 - todayBf).toFixed(1);
  const ffmKg = +(todayWeight * (ffmPct / 100.0)).toFixed(1);
  const musclePct = +(ffmPct * 0.6468).toFixed(1);
  const waterPct = +(ffmPct * 0.7222).toFixed(1);
  const visceral = Math.max(1, Math.round(11 + (todayBf - 24.4) * 0.48));
  const tmbKcal = extractedDay.tmb_kcal || Math.round(370 + (21.6 * ffmKg));

  const prevSleepTotal = parseInt(existing.total_sleep_min, 10) || 0;
  const newSleepTotal = parseInt(extractedDay.sleep_total_min, 10) || 0;
  const useNewSleep = newSleepTotal >= prevSleepTotal && newSleepTotal >= 120;
  const napNew = Object.keys(extractedDay.nap_map || {}).reduce((a, k) => a + extractedDay.nap_map[k], 0);
  const napTotal = Math.max(parseInt(existing.nap_sleep_min, 10) || 0, napNew);

  const avg = arr => arr.reduce((a, b) => a + b, 0) / arr.length;
  const median = arr => {
    if (!arr.length) return 0;
    const sorted = [...arr].sort((a, b) => a - b);
    const mid = Math.floor(sorted.length / 2);
    return sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
  };
  const mergedRaw = {
    log_date: dateStr,
    weight_kg: todayWeight,
    body_fat_pct: todayBf,
    skeletal_muscle_pct: musclePct,
    visceral_fat_index: visceral,
    tmb_kcal: tmbKcal,
    body_water_pct: waterPct,
    rmssd_ms: extractedDay.rmssd_list.length ? +median(extractedDay.rmssd_list).toFixed(1) : (existing.rmssd_ms || 0),
    rhr_bpm: extractedDay.rhr_list.length ? +median(extractedDay.rhr_list).toFixed(1) : (existing.rhr_bpm || 0),
    resp_rate_brpm: extractedDay.resp_list.length ? +avg(extractedDay.resp_list).toFixed(1) : (existing.resp_rate_brpm || 0),
    spo2_pct: extractedDay.spo2_list.length ? +avg(extractedDay.spo2_list).toFixed(1) : (existing.spo2_pct || 0),
    total_sleep_min: useNewSleep ? newSleepTotal : (prevSleepTotal || 0),
    time_in_bed_min: useNewSleep ? (extractedDay.sleep_tib_min || Math.round(newSleepTotal * 1.05)) : (existing.time_in_bed_min || 0),
    deep_sleep_min: useNewSleep ? (extractedDay.sleep_deep_min || 0) : (existing.deep_sleep_min || 0),
    rem_sleep_min: useNewSleep ? (extractedDay.sleep_rem_min || 0) : (existing.rem_sleep_min || 0),
    light_sleep_min: useNewSleep ? (extractedDay.sleep_light_min || 0) : (existing.light_sleep_min || 0),
    steps_count: todaySteps,
    nap_sleep_min: napTotal,
    pickleball_or_gym_minutes: todayWorkoutMin,
    session_rpe: todayWorkoutMin > 0 ? (existing.session_rpe || 7.0) : 0,
    is_hospital_shift: Boolean(existing.is_hospital_shift),
    night_awake_shift_hours: parseFloat(existing.night_awake_shift_hours) || 0,
    intake_kcal: totalKcal,
    intake_protein_g: totalProt,
    intake_carbs_g: totalCarbs,
    intake_fat_g: totalFat,
    meals_log_json: mealsList
  };

  supabaseUpsert_('daily_raw_biometrics', mergedRaw);

  const prevCompRows = supabaseGet_(`computed_daily_scores?log_date=lt.${dateStr}&order=log_date.desc&limit=1`);
  const prevComp = prevCompRows.length > 0 ? prevCompRows[0] : null;

  const computedScore = computeArgonMbeScore_(mergedRaw, historyRows, prevComp);
  supabaseUpsert_('computed_daily_scores', computedScore);

  return { ...mergedRaw, ...computedScore };
}

// ===ENGINE_START=== (copiado automáticamente de index.html; no editar a mano)
function computeArgonEngine_(todayRaw, historyRawList, prevComputed) {
      const wakeTimeStr = ARGON_CONFIG.HABITUAL_WAKE_TIME;
      const MIN_BASELINE_DAYS = 7;
      const basalNeedH = 7.50; // Calibrado a 7h 30m de sueño neto efectivo

      // Edad cronológica exacta (Nacido: 12 Nov 1997)
      const birthDate = new Date('1997-11-12T00:00:00-04:00');
      const refDate = todayRaw.log_date ? new Date(todayRaw.log_date + 'T12:00:00-04:00') : new Date();
      const chronologicalAge = +((refDate.getTime() - birthDate.getTime()) / (365.2425 * 86400000)).toFixed(2);
      const heightM = 1.73;

      // A. BIOIMPEDANCIA FEELFIT
      const rawW = parseFloat(todayRaw.weight_kg) || (prevComputed ? prevComputed.weight_kg : 86.80);
      const rawBf = parseFloat(todayRaw.body_fat_pct) || (prevComputed ? prevComputed.body_fat_pct : 24.1);
      const ffmPct = +(100.0 - rawBf).toFixed(1);

      const fatMassKg = +(rawW * (rawBf / 100.0)).toFixed(2);
      const ffmKg = +(rawW - fatMassKg).toFixed(1);
      const boneMassKg = +(rawW * 0.0378).toFixed(1);
      const muscleMassKg = +(ffmKg - boneMassKg).toFixed(1);

      const musclePct = parseFloat(todayRaw.skeletal_muscle_pct) || +(ffmPct * 0.6468).toFixed(1);
      const waterPct = parseFloat(todayRaw.body_water_pct) || +(ffmPct * 0.7222).toFixed(1);
      const visceral = parseFloat(todayRaw.visceral_fat_index) || Math.max(1, Math.round(11 + (rawBf - 24.4) * 0.48));
      const tmbFeelfit = parseInt(todayRaw.tmb_kcal, 10) || Math.round(370 + (21.6 * ffmKg));
      const waterMassKg = +(rawW * (waterPct / 100.0)).toFixed(2);
      const bmi = +(rawW / (heightM * heightM)).toFixed(1);

      const prevTrendW = prevComputed && prevComputed.trend_weight_kg ? parseFloat(prevComputed.trend_weight_kg) : rawW;
      const prevTrendBf = prevComputed && prevComputed.trend_body_fat_pct ? parseFloat(prevComputed.trend_body_fat_pct) : rawBf;
      const trendW = +((0.25 * rawW) + (0.75 * prevTrendW)).toFixed(2);
      const trendBf = +((0.25 * rawBf) + (0.75 * prevTrendBf)).toFixed(2);

      // B. STRAIN 4D
      const steps = Math.max(0, parseInt(todayRaw.steps_count, 10) || 0);
      const workoutMin = Math.max(0, parseFloat(todayRaw.pickleball_or_gym_minutes) || 0);
      const sRpe = Math.max(0, parseFloat(todayRaw.session_rpe) || (workoutMin > 0 ? 7.0 : 0));
      const nightAwakeH = Math.max(0, parseFloat(todayRaw.night_awake_shift_hours) || 0);

      // Pasos netos (descuento pasos deportivos ~85 pasos/min ejercicio) para NEAT
      const stepsNet = Math.max(0, steps - workoutMin * 85);

      const ambientZ1 = Math.round(Math.max(0, stepsNet - 2000) / 320);

      // Zonas dinámicas por RPE cuando no hay hr_zones_real
      let zones;
      if (todayRaw.hr_zones_real === true && todayRaw.hr_zones_minutes) {
        zones = todayRaw.hr_zones_minutes;
      } else {
        // Distribución base
        let baseZ1 = ambientZ1 + Math.round(workoutMin * 0.22);
        let baseZ2 = Math.round(workoutMin * 0.33);
        let baseZ3 = Math.round(workoutMin * 0.20);
        let baseZ4 = Math.round(workoutMin * 0.17);
        let baseZ5 = Math.round(workoutMin * 0.08);

        // Redistribución Z1-Z2 -> Z3-Z5 proporcional al RPE (RPE 1-10, centro 5.5)
        if (workoutMin > 0 && sRpe > 0) {
          const rpeFactor = Math.max(0, Math.min(1, (sRpe - 1) / 9)); // 0-1
          const shiftAmount = (baseZ1 + baseZ2) * rpeFactor * 0.5; // máx 50% del peso bajo
          baseZ1 = Math.round(baseZ1 * (1 - rpeFactor * 0.5));
          baseZ2 = Math.round(baseZ2 * (1 - rpeFactor * 0.5));
          const totalHigh = baseZ3 + baseZ4 + baseZ5;
          if (totalHigh > 0) {
            baseZ3 += Math.round(shiftAmount * (baseZ3 / totalHigh));
            baseZ4 += Math.round(shiftAmount * (baseZ4 / totalHigh));
            baseZ5 += Math.round(shiftAmount * (baseZ5 / totalHigh));
          } else {
            baseZ3 += Math.round(shiftAmount * 0.5);
            baseZ4 += Math.round(shiftAmount * 0.3);
            baseZ5 += Math.round(shiftAmount * 0.2);
          }
        }
        zones = { z1: baseZ1, z2: baseZ2, z3: baseZ3, z4: baseZ4, z5: baseZ5 };
      }

      const z13Min = (parseFloat(zones.z1) || 0) + (parseFloat(zones.z2) || 0) + (parseFloat(zones.z3) || 0);
      const z45Min = (parseFloat(zones.z4) || 0) + (parseFloat(zones.z5) || 0);

      const mids = { z1: 0.55, z2: 0.65, z3: 0.75, z4: 0.85, z5: 0.95 };
      let cardioTrimp = 0;
      for (const z in mids) {
        cardioTrimp += (parseFloat(zones[z]) || 0) * mids[z] * 0.64 * Math.exp(1.92 * mids[z]);
      }
      // Eliminada duplicidad: pasos basales ya inyectados en zones.z1 vía ambientZ1

      // neuroLoad: solo componente neuromuscular real (sRPE * tiempo), sin duplicar Z4-Z5
      const neuroLoad = 0.85 * (sRpe * workoutMin);
      const shiftLoad = nightAwakeH > 0 ? (1.40 * nightAwakeH * 35.0) : 0.0;
      const totalLoad = cardioTrimp + neuroLoad + shiftLoad;
      const strainScore = Math.min(20.8, Math.max(0.0, 21.0 * (1.0 - Math.exp(-0.00165 * totalLoad))));

      // C. SUEÑO EN 4 PILARES + AMORTIZACIÓN DE DEUDA (PROCESO S BORBÉLY)
      const prevTst = prevComputed && prevComputed.total_sleep_min ? prevComputed.total_sleep_min : 450;
      const sleepMissing = !(parseInt(todayRaw.total_sleep_min, 10) >= 120) || (parseInt(todayRaw.total_sleep_min, 10) === 450 && parseInt(todayRaw.deep_sleep_min, 10) === 84 && parseInt(todayRaw.rem_sleep_min, 10) === 96);
      const tstMin = sleepMissing ? 0 : parseInt(todayRaw.total_sleep_min, 10);
      // Eficiencia basal realista: Math.round(tstMin / 0.87) en vez de tstMin * 1.06
      const tibMin = sleepMissing ? 0 : (parseInt(todayRaw.time_in_bed_min, 10) || Math.max(tstMin + 25, Math.round(tstMin / 0.87)));
      const deepMin = sleepMissing ? 0 : (parseInt(todayRaw.deep_sleep_min, 10) || Math.round(tstMin * 0.20));
      const remMin = sleepMissing ? 0 : (parseInt(todayRaw.rem_sleep_min, 10) || Math.round(tstMin * 0.22));
      const lightMin = Math.max(0, tstMin - deepMin - remMin);
      const napMin = parseInt(todayRaw.nap_sleep_min, 10) || 0;

      const sePct = Math.min(100, (tstMin / Math.max(1, tibMin)) * 100);
      const swsPct = (deepMin / Math.max(1, tstMin)) * 100;
      const remPct = (remMin / Math.max(1, tstMin)) * 100;

      // Limpieza de noches vacías: solo noches válidas (total_sleep_min >= 120) para deuda y regularidad
      const validSleeps = [todayRaw, ...historyRawList.slice(0, 6)].filter(r => parseInt(r.total_sleep_min, 10) >= 120);

      validSleeps.forEach((r, idx) => {
        const pastTstH = (parseInt(r.total_sleep_min, 10) + (parseInt(r.nap_sleep_min, 10) || 0)) / 60.0;
        weightedDebtH += (basalNeedH - pastTstH) * Math.exp(-0.35 * idx);
      });
      weightedDebtH = Math.max(0, +weightedDebtH.toFixed(2));

      const dynamicNeedH = Math.min(10.0, Math.max(6.5, basalNeedH + (0.042 * strainScore) + (0.35 * weightedDebtH)));
      const prevStrain = prevComputed && parseFloat(prevComputed.strain_score) ? parseFloat(prevComputed.strain_score) : 0;
      const needForRecoveryH = Math.min(10.0, Math.max(6.5, basalNeedH + (0.042 * prevStrain) + (0.35 * weightedDebtH)));
      const hoursVsNeededPct = Math.min(100, Math.round((((tstMin + napMin) / 60.0) / needForRecoveryH) * 100));

      const recentValidTst = validSleeps.map(r => parseInt(r.total_sleep_min, 10));
      const meanTst = recentValidTst.length ? recentValidTst.reduce((a, b) => a + b, 0) / recentValidTst.length : 450;
      const sdTstHours = recentValidTst.length > 1 ? Math.sqrt(recentValidTst.reduce((s, v) => s + Math.pow((v - meanTst) / 60.0, 2), 0) / recentValidTst.length) : 0;
      const sleepRegularityPct = Math.min(98, Math.max(55, Math.round(94 - (sdTstHours * 14.0) - (nightAwakeH * 3.5))));

      // Rebalanceo de etapas PPG: 0.50 * hoursVsNeededPct + 0.25 * sleepRegularityPct + 0.15 * seScore + 0.10 * stagesScore
      const sleepArchScore = Math.round(
        0.50 * hoursVsNeededPct +
        0.25 * sleepRegularityPct +
        0.15 * Math.min(100, (sePct / 88.0) * 100) +
        0.10 * Math.min(100, ((swsPct + remPct) / 40.0) * 100)
      );

      const [wakeH, wakeM] = wakeTimeStr.split(':').map(Number);
      const reqTibHours = dynamicNeedH / Math.max(0.80, sePct / 100.0);
      const wakeTotalMin = (wakeH * 60) + wakeM + 1440;
      const startMin = Math.round(wakeTotalMin - (reqTibHours * 60) - 15) % 1440;
      const endMin = Math.round(wakeTotalMin - (reqTibHours * 60) + 15) % 1440;

      const rawRmssdCandidate = parseFloat(todayRaw.rmssd_ms) || prevRmssd;

        .filter(r => parseFloat(r.rmssd_ms) > 0 && !(parseFloat(r.rmssd_ms) === 50 && parseFloat(r.rhr_bpm) === 54.5))
        .map(r => parseFloat(r.rmssd_ms) || 49.0);

      const rmssdToday = rawRmssdCandidate > dynamicRmssdCap ? prevRmssd : rawRmssdCandidate;

      const rhrToday = parseFloat(todayRaw.rhr_bpm) || prevRhr;
      const respToday = parseFloat(todayRaw.resp_rate_brpm) || 14.1;
      const spo2Today = parseFloat(todayRaw.spo2_pct) || 97.6;

      const lnRmssdToday = Math.log(Math.max(5.0, rmssdToday));
      const histRmssd = histRmssdRaw.map(v => v > dynamicRmssdCap ? meanRmssd14 : v);
      const histLn = histRmssd.map(v => Math.log(Math.max(5.0, v)));
      const histRhr = historyRawList.slice(0, 14).filter(r => parseFloat(r.rhr_bpm) > 0 && !(parseFloat(r.rmssd_ms) === 50 && parseFloat(r.rhr_bpm) === 54.5)).map(r => parseFloat(r.rhr_bpm));
      const histResp = historyRawList.slice(0, 14).filter(r => parseFloat(r.resp_rate_brpm) > 0).map(r => parseFloat(r.resp_rate_brpm));

      const meanArr = (arr) => arr.length ? arr.reduce((a, b) => a + b, 0) / arr.length : 0;
      const sdArr = (arr, m) => arr.length > 1 ? Math.sqrt(arr.reduce((s, v) => s + Math.pow(v - m, 2), 0) / arr.length) : 0.12;

      const meanRmssd = histRmssd.length >= MIN_BASELINE_DAYS ? meanArr(histRmssd) : rmssdToday;
      const meanLn = histLn.length >= MIN_BASELINE_DAYS ? meanArr(histLn) : lnRmssdToday;
      // Piso de dispersión reducido: 0.04 en vez de 0.08
      const sdLn = Math.max(0.04, histLn.length >= MIN_BASELINE_DAYS ? sdArr(histLn, meanLn) : 0.12);
      const meanRhr = histRhr.length >= MIN_BASELINE_DAYS ? meanArr(histRhr) : rhrToday;
      const sdRhr = Math.max(1.5, histRhr.length >= MIN_BASELINE_DAYS ? sdArr(histRhr, meanRhr) : 2.2);
      const meanResp = histResp.length >= MIN_BASELINE_DAYS ? meanArr(histResp) : 14.2;

      const histLn7 = histLn.slice(0, 7);
      const cvLnRmssd7 = histLn7.length >= 3 ? (sdArr(histLn7, meanArr(histLn7)) / Math.max(0.001, meanArr(histLn7))) * 100 : 0;
      const swc = 0.5 * sdLn; // Smallest Worthwhile Change

      const hrvMissing = !(parseFloat(todayRaw.rmssd_ms) > 0) || (parseFloat(todayRaw.rmssd_ms) === 50 && parseFloat(todayRaw.rhr_bpm) === 54.5);
      const hrvZ = hrvMissing ? 0 : Math.max(-3, Math.min(3, (lnRmssdToday - meanLn) / sdLn));
      const rhrMissing = !(parseFloat(todayRaw.rhr_bpm) > 0) || (parseFloat(todayRaw.rmssd_ms) === 50 && parseFloat(todayRaw.rhr_bpm) === 54.5);
      const rhrZ = rhrMissing ? 0 : Math.max(-3, Math.min(3, (rhrToday - meanRhr) / sdRhr));
      const penalty = (spo2Today < 94.0 ? 1.0 : 0.0) + (respToday > 16.8 ? 0.85 : 0.0);

      // Estrés nocturno puro: solo rhrZ y hrvZ (eliminado término SWS+REM)
      const highSleepStressPct = Math.min(45, Math.max(0, Math.round(
        Math.max(0, rhrZ * 9.0) + Math.max(0, -hrvZ * 7.0)
      )));

      const sleepZ = sleepMissing ? 0 : Math.max(-3, Math.min(2, (sleepArchScore - 92.0) / 10.0));
      // Amortiguación psi con SWC: reduce impacto si cambio < SWC
      const psiRaw = (0.55 * hrvZ) - (0.20 * rhrZ) + (0.25 * sleepZ) - penalty;
      const psi = Math.abs(psiRaw) < swc ? psiRaw * 0.5 : psiRaw;
      const recoveryScore = Math.min(99, Math.max(1, Math.round(100.0 / (1.0 + Math.exp(-1.15 * psi)))));

      // ACWR: carga aguda 3d vs crónica 14d -> ajuste recommended_max_strain
      let acwrPenalty = 0;
      if (historyRawList.length >= 14) {
        const acuteLoad = historyRawList.slice(0, 3).reduce((sum, r) => sum + (parseFloat(r.strain_score) || 0), 0) / 3;
        const chronicLoad = historyRawList.slice(3, 14).reduce((sum, r) => sum + (parseFloat(r.strain_score) || 0), 0) / 11;
        const acwr = chronicLoad > 0 ? acuteLoad / chronicLoad : 1;
        if (acwr > 1.3) acwrPenalty = Math.min(2.0, (acwr - 1.3) * 5); // máx -2.0 puntos
      }

      let recStatus = 'MODERATE';
      let recMaxStrain = 12.0;
      if (recoveryScore >= 67) {
        recStatus = 'OPTIMAL';
        recMaxStrain = +(14.5 + (recoveryScore - 67) * 0.14).toFixed(1);
      } else if (recoveryScore < 34) {
        recStatus = 'COMPROMISED';
        recMaxStrain = +(6.0 + recoveryScore * 0.10).toFixed(1);
      } else {
        recMaxStrain = +(10.0 + (recoveryScore - 34) * 0.13).toFixed(1);
      }
      // Aplicar penalización ACWR
      recMaxStrain = Math.max(4.0, +(recMaxStrain - acwrPenalty).toFixed(1));

      // E. NUTRICIÓN, BITÁCORA DE COMIDAS Y LEY DE ATWATER (0 KCAL DESVIACIÓN)
      // NEAT usa stepsNet (pasos no deportivos)
      const neatStepsKcal = Math.max(0, (stepsNet - 2500) * 0.035);
      const netExerciseKcal = workoutMin * (strainScore >= 13 ? 5.5 : 4.2);

      // TEF exacto por macronutriente (Atwater): proteína 25%, carbs 8%, grasa 2%
      const targetProt = Math.round(2.35 * ffmKg);
      const targetFat = Math.round(0.72 * rawW);
      const targetCarbs = Math.max(45, Math.round((0 - (targetProt * 4) - (targetFat * 9)) / 4)); // temporal, recalculado abajo

      // TEF = (TMB * 1.05) + (prot_g * 4 * 0.25) + (carbs_g * 4 * 0.08) + (fat_g * 9 * 0.02)
      // Iteramos una vez para estimar TEF con macros objetivo
      const tefEstimate = (tmbFeelfit * 1.05) + (targetProt * 4 * 0.25) + (targetCarbs * 4 * 0.08) + (targetFat * 9 * 0.02);
      const rmrActive = tefEstimate;

      // TDEE adaptativo 14d: blend 50% teórico + 50% empírico si hay ≥7 días con intake ≥1200
      let dynamicTdee = Math.round(rmrActive + neatStepsKcal + netExerciseKcal);
      const validIntakeDays = historyRawList.slice(0, 14).filter(r => parseInt(r.intake_kcal, 10) >= 1200);
      if (validIntakeDays.length >= 7) {
        const meanIntake = validIntakeDays.reduce((s, r) => s + parseInt(r.intake_kcal, 10), 0) / validIntakeDays.length;
        const trendW = parseFloat(prevComputed?.trend_weight_kg || todayRaw.weight_kg || rawW);
        const prevTrendW = historyRawList.length >= 14 ? parseFloat(historyRawList[13].weight_kg || trendW) : trendW;
        const deltaTrendW = trendW - prevTrendW;
        const empiricalTdee = meanIntake - (deltaTrendW * 7700 / validIntakeDays.length);
        dynamicTdee = Math.round((dynamicTdee + empiricalTdee) / 2);
      }

      const desiredDeficit = strainScore >= 14.0 ? 500 : 550;
      const rawTargetKcal = Math.min(2450, Math.max(1650, dynamicTdee - desiredDeficit));

      const targetCarbsFinal = Math.max(45, Math.round((rawTargetKcal - (targetProt * 4) - (targetFat * 9)) / 4));
      const targetKcal = (targetProt * 4) + (targetCarbsFinal * 4) + (targetFat * 9);

      const mealsLog = Array.isArray(todayRaw.meals_log_json) ? todayRaw.meals_log_json : [];
      let realIntakeKcal = Math.max(0, parseInt(todayRaw.intake_kcal, 10) || 0);
      let realProt = Math.max(0, parseFloat(todayRaw.intake_protein_g) || 0);
      let realCarbs = Math.max(0, parseFloat(todayRaw.intake_carbs_g) || 0);
      let realFat = Math.max(0, parseFloat(todayRaw.intake_fat_g) || 0);

function logWebhookAudit_(payload, detectedDates, syncedSummaries) {
  const topKeys = (payload && typeof payload === 'object' && !Array.isArray(payload)) ? Object.keys(payload) : ['array'];
  const light = Array.isArray(payload) ? payload : Object.assign({}, payload);
  if (!Array.isArray(light)) ['heart_rate', 'total_calories', 'active_calories', 'distance'].forEach(k => { delete light[k]; });
  const rawSampleStr = JSON.stringify(light).substring(0, 60000);
  let rawSampleObj = {};
  try { rawSampleObj = JSON.parse(rawSampleStr); } catch (_) { rawSampleObj = { preview: rawSampleStr }; }

  supabaseInsert_('webhook_logs', {
    source_app: 'HC_WEBHOOK',
    detected_dates: detectedDates,
    detected_categories: topKeys,
    parsed_summary: syncedSummaries,
    raw_payload: rawSampleObj
  });
}



function sbHeaders_(extra) {
  const k = ARGON_CONFIG.SUPABASE_KEY;
  return Object.assign({ apikey: k, Authorization: 'Bearer ' + k, 'Content-Type': 'application/json' }, extra || {});
}

function supabaseGet_(path) {
  const res = UrlFetchApp.fetch(`${ARGON_CONFIG.SUPABASE_URL}/rest/v1/${path}`, {
    method: 'get', headers: sbHeaders_(), muteHttpExceptions: true
  });
  const body = JSON.parse(res.getContentText() || '[]');
  if (res.getResponseCode() >= 300 || !Array.isArray(body)) {
    throw new Error(`Supabase GET ${res.getResponseCode()}: ${res.getContentText().slice(0, 200)}`);
  }
  return body;
}

function sbWrite_(table, obj, merge) {
  const res = UrlFetchApp.fetch(`${ARGON_CONFIG.SUPABASE_URL}/rest/v1/${table}`, {
    method: 'post',
    headers: sbHeaders_({ Prefer: (merge ? 'resolution=merge-duplicates,' : '') + 'return=minimal' }),
    payload: JSON.stringify(obj),
    muteHttpExceptions: true
  });
  if (res.getResponseCode() >= 300) {
    throw new Error(`Supabase ${table} ${res.getResponseCode()}: ${res.getContentText().slice(0, 200)}`);
  }
}
const supabaseUpsert_ = (t, o) => sbWrite_(t, o, true);
const supabaseInsert_ = (t, o) => sbWrite_(t, o, false);

function jsonResponse_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

      snapshot_date: todayComp.log_date
    });
  } catch (err) {
    return jsonResponse_({ status: 'error', reply: 'Error interno Coach: ' + err.message });
  }
}

function repairMealsFromLogs_() {
  const logs = supabaseGet_('webhook_logs?select=raw_payload&order=received_at.desc&limit=60');
  const seen = {}, items = [];
  logs.forEach(l => {
    const p = l.raw_payload;
    if (!p || !Array.isArray(p.nutrition)) return;
    p.nutrition.forEach(n => {
      const k = [n.name, n.calories, n.start_time].join('|');
      if (!seen[k]) { seen[k] = 1; items.push(n); }
    });
  });
  const days = extractHealthConnectByDate_({ nutrition: items });
  const dates = Object.keys(days).filter(d => Object.keys(days[d].meals_map).length).sort();
  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    dates.forEach(d => upsertAndComputeDayInSupabase_(d, days[d]));
  } finally {
    lock.releaseLock();
  }
  Logger.log('Comidas reparadas en ' + dates.length + ' fecha(s): ' + dates.join(', ') + ' | items unicos: ' + items.length);
}

function rebuildMealsFromLogs_(dateStr) {
  dateStr = dateStr || Utilities.formatDate(new Date(), ARGON_CONFIG.TIMEZONE, 'yyyy-MM-dd');
  const logs = supabaseGet_('webhook_logs?select=raw_payload&order=received_at.desc&limit=100');
  const best = {};
  let logsConNutricion = 0, logsTruncados = 0;
  logs.forEach(l => {
    const p = l.raw_payload;
    if (!p || p.preview) { logsTruncados++; return; }
    if (!Array.isArray(p.nutrition)) return;
    logsConNutricion++;
    const cnt = {}, ref = {};
    p.nutrition.forEach(n => {
      const k = [n.name, n.calories, n.start_time].join('|');
      cnt[k] = (cnt[k] || 0) + 1;
      ref[k] = n;
    });
    Object.keys(cnt).forEach(k => { if (!best[k] || cnt[k] > best[k].c) best[k] = { c: cnt[k], n: ref[k] }; });
  });
  const items = [];
  Object.keys(best).forEach(k => { for (let i = 0; i < best[k].c; i++) items.push(best[k].n); });
  const day = extractHealthConnectByDate_({ nutrition: items })[dateStr];
  const meals = day ? Object.keys(day.meals_map).map(k => day.meals_map[k]) : [];
  Logger.log('Logs: ' + logs.length + ' | con nutricion: ' + logsConNutricion + ' | truncados/sin datos: ' + logsTruncados + ' | items unicos: ' + items.length + ' | comidas del ' + dateStr + ': ' + meals.length);
  if (!meals.length) { Logger.log('No hay comidas de ' + dateStr + ' recuperables en los logs. No se modifico nada.'); return; }

  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    const rows = supabaseGet_('daily_raw_biometrics?log_date=eq.' + dateStr + '&limit=1');
    const manual = (rows.length && Array.isArray(rows[0].meals_log_json) ? rows[0].meals_log_json : [])
      .filter(m => m && String(m.id || '').indexOf('manual_') === 0);
    supabaseUpsert_('daily_raw_biometrics', { log_date: dateStr, meals_log_json: manual, intake_kcal: 0, intake_protein_g: 0, intake_carbs_g: 0, intake_fat_g: 0 });
    const out = upsertAndComputeDayInSupabase_(dateStr, day);
    Logger.log('RESULTADO ' + dateStr + ': ' + out.intake_kcal + ' kcal | P ' + out.intake_protein_g + ' | C ' + out.intake_carbs_g + ' | G ' + out.intake_fat_g + ' | comidas: ' + (out.meals_log_json || []).length);
  } finally {
    lock.releaseLock();
  }
}

function rebuildMealsHoy_() { rebuildMealsFromLogs_(); }
function rebuildMealsAyer_() {
  rebuildMealsFromLogs_(Utilities.formatDate(new Date(Date.now() - 86400000), ARGON_CONFIG.TIMEZONE, 'yyyy-MM-dd'));
}

function repairSleepFromLogs_() {
  const logs = supabaseGet_('webhook_logs?select=raw_payload&order=received_at.desc&limit=100');
  const seen = {}, items = [];
  logs.forEach(l => {
    const p = l.raw_payload;
    if (!p || !Array.isArray(p.sleep)) return;
    p.sleep.forEach(s => {
      const k = String(s.session_end_time || s.end_time || '') + '|' + (s.duration_seconds || '');
      if (!seen[k]) { seen[k] = 1; items.push(s); }
    });
  });
  const days = extractHealthConnectByDate_({ sleep: items });
  const dates = Object.keys(days).filter(d => days[d].sleep_total_min).sort();
  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    dates.forEach(d => {
      const o = upsertAndComputeDayInSupabase_(d, days[d]);
      Logger.log(d + ': sueno ' + o.total_sleep_min + ' min (profundo ' + o.deep_sleep_min + ', REM ' + o.rem_sleep_min + ')');
    });
  } finally {
    lock.releaseLock();
  }
  Logger.log('Sueno reparado en ' + dates.length + ' fecha(s) | sesiones unicas: ' + items.length);
}
